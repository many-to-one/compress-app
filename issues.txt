# GIT COMMANDS
# Jeśli nie masz lokalnych zmian, które chcesz zachować:
git fetch origin
git reset --hard origin/main

# Jeśli masz lokalne zmiany i chcesz je połączyć:
git pull --no-rebase origin main

# zobaczyć wszystkie commity
git log --oneline --graph --decorate --all

# Jak przełączyć się na konkretny commit
git checkout <hash>

# Utworzyć nową gałąź
git checkout -b nowa-galaz <hash>

# Wrócić na gałąź main
git checkout main

# Przenieść commit z main na v1
# Będąc na gałęzi v1:
git cherry-pick e5db671
git push origin v1
